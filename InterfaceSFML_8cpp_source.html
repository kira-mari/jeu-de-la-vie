68void InterfaceSFML::mettreAJourVue() {
69 const Grille& grille = jeu.obtenirGrille();
70 float worldWidth = static_cast<float>(grille.obtenirNbColonnes() * tailleCellule);
71 float worldHeight = static_cast<float>(grille.obtenirNbLignes() * tailleCellule);
72
73 // Définir la taille et le centre de la vue en coordonnées monde
74 #ifdef SFML_VERSION_3
75 vue.setSize({worldWidth, worldHeight});
76 vue.setCenter({worldWidth / 2.f, worldHeight / 2.f});
77 #else
78 vue.setSize(worldWidth, worldHeight);
79 vue.setCenter(worldWidth / 2.f, worldHeight / 2.f);
80 #endif
81
82 // Calculer le viewport (lettres-boxing) pour préserver l'aspect
83 sf::Vector2u winSize = fenetre->getSize();
84 float winW = static_cast<float>(winSize.x);
85 float winH = static_cast<float>(winSize.y);
86 if (winW <= 0.f || winH <= 0.f) return;
87
88 float windowAspect = winW / winH;
89 float worldAspect = (worldWidth > 0.f && worldHeight > 0.f) ? (worldWidth / worldHeight) : 1.f;
90
91 float viewportLeft = 0.f, viewportTop = 0.f, viewportW = 1.f, viewportH = 1.f;
92 if (windowAspect > worldAspect) {
93 // fenetre plus large -> barres horizontales
94 viewportW = worldAspect / windowAspect;
95 viewportLeft = (1.f - viewportW) / 2.f;
96 viewportTop = 0.f;
97 viewportH = 1.f;
98 } else {
99 // fenetre plus haute -> barres verticales
100 viewportH = windowAspect / worldAspect;
101 viewportTop = (1.f - viewportH) / 2.f;
102 viewportLeft = 0.f;
103 viewportW = 1.f;
104 }
105
106 // Construct viewport as position+size to be compatible across SFML versions
107 sf::FloatRect viewport(sf::Vector2f(viewportLeft, viewportTop), sf::Vector2f(viewportW, viewportH));
108 vue.setViewport(viewport);
109}

482// Implémentation du zoom centré sur un pixel (définie après gererPlacementMotif pour éviter insertion accidentelle)
483void InterfaceSFML::zoomerSurPixel(const sf::Vector2i& pixel, float factor) {
484 // Conserver la position monde du pixel, appliquer zoom, recentrer pour que
485 // le pixel pointe vers le même point monde après zoom
486 auto avant = fenetre->mapPixelToCoords(pixel);
487 // Appliquer zoom à la vue
488 vue.zoom(factor);
489 // Mettre la vue dans la fenêtre avant de calculer les nouvelles coordonnées
490 fenetre->setView(vue);
491 // Clamper niveau logique (après application)
492 zoomNiveau *= (1.0f / factor);
493 if (zoomNiveau < zoomMin) zoomNiveau = zoomMin;
494 if (zoomNiveau > zoomMax) zoomNiveau = zoomMax;
495 auto apres = fenetre->mapPixelToCoords(pixel);
496 sf::Vector2f decalage = avant - apres;
497 auto centre = vue.getCenter();
498 vue.setCenter(centre + decalage);
499 fenetre->setView(vue);
500}

509void InterfaceSFML::dessinerGrille() {
510 const Grille& grille = jeu.obtenirGrille();
511
512 #ifdef SFML_VERSION_3
513 sf::RectangleShape celluleRect({static_cast<float>(tailleCellule - 1), static_cast<float>(tailleCellule - 1)});
514 #else
515 sf::RectangleShape celluleRect(sf::Vector2f(static_cast<float>(tailleCellule - 1), static_cast<float>(tailleCellule - 1)));
516 #endif
517
518 for (int ligne = 0; ligne < grille.obtenirNbLignes(); ++ligne) {
519 for (int colonne = 0; colonne < grille.obtenirNbColonnes(); ++colonne) {
520 const Cellule& cellule = grille.obtenirCellule(ligne, colonne);
521
522 // Choisir la couleur selon l'état
523 if (cellule.estObstacle()) {
524 if (cellule.estVivante()) {
525 // Obstacle vivant = ORANGE
526 celluleRect.setFillColor(sf::Color(
527 Config::CouleurObstacleVivantSFML::R,
528 Config::CouleurObstacleVivantSFML::G,
529 Config::CouleurObstacleVivantSFML::B
530 ));
531 } else {
532 // Obstacle mort = ROUGE
